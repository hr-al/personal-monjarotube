cmake_minimum_required(VERSION 3.22.1)
project(monjarotube C)
file(GLOB BYEDPI_SOURCES CONFIGURE_DEPENDS "byedpi/*.c")
list(REMOVE_ITEM BYEDPI_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/byedpi/win_service.c")
add_library(monjarodpi SHARED ${BYEDPI_SOURCES} bridge.c)
target_include_directories(monjarodpi PRIVATE byedpi)
target_compile_features(monjarodpi PRIVATE c_std_11)
target_compile_definitions(monjarodpi PRIVATE ANDROID_APP _XOPEN_SOURCE=500)
target_compile_options(monjarodpi PRIVATE -O2)
target_link_options(monjarodpi PRIVATE "-Wl,-z,max-page-size=16384")
target_link_libraries(monjarodpi PRIVATE log)
